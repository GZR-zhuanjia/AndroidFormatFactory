cmake_minimum_required(VERSION 3.22.1)
project(converter_audio LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(NATIVE_DEPS "${CMAKE_CURRENT_SOURCE_DIR}/../../../../native-deps/${ANDROID_ABI}" ABSOLUTE)
foreach(component avformat avcodec swresample avutil mp3lame)
    set(library "${NATIVE_DEPS}/lib/lib${component}.so")
    if(NOT EXISTS "${library}")
        message(FATAL_ERROR "Missing ${library}. Run ./scripts/build-native.sh --abis ${ANDROID_ABI} with Android NDK 28.2.13676358 first.")
    endif()
    add_library(${component} SHARED IMPORTED)
    set_target_properties(${component} PROPERTIES IMPORTED_LOCATION "${library}")
endforeach()
add_library(converter_audio SHARED native_audio.cpp)
target_include_directories(converter_audio PRIVATE "${NATIVE_DEPS}/include")
target_compile_options(converter_audio PRIVATE -Wall -Wextra -Werror=return-type)
target_link_options(converter_audio PRIVATE -Wl,-z,max-page-size=16384)
target_link_libraries(converter_audio PRIVATE avformat avcodec swresample avutil mp3lame log)
